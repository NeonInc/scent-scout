                return;

            var maxStock = parseInt(availableStock, 10);

                quantityInput.value = currentVal;

                quantityInput.value = 1;

            if (!quantityInput) {

            var currentVal = parseInt(quantityInput.value, 10);
            if (!isNaN(currentVal)) {
                quantityInput.value = currentVal > 1 ? currentVal - 1 : 1;
            } else {
                quantityInput.value = 1;
            }

                quantityInput.value = 1;
                return;
            }
            quantityInput.value = Math.min(currentVal, Math.min(10, maxStock));

            function updateFavorites(product) {
                var favoritesData = product.Favorite || product.favorite;
                if (!favoritesData || !favoritesData.length) {
                    return;
                }
                var favoriteState = favoritesData[0] || {};
                var isInAnyFavoriteList = favoriteState.isInAnyFavoriteList === true;
                var favoriteLists = favoriteState.FavoriteLists || favoriteState.favoriteLists || [];
                Array.prototype.slice.call(document.querySelectorAll(".js-favorite-btn")).forEach(function (container) {
                    var mainIcon = container.querySelector("label i");
                    if (mainIcon) {
                        mainIcon.classList.toggle("fas", isInAnyFavoriteList);
                        mainIcon.classList.toggle("fal", !isInAnyFavoriteList);
                    }
                    var actionButtons = container.querySelectorAll(".js-fav-action");
                    Array.prototype.slice.call(actionButtons).forEach(function (button, index) {
                        var listItem = favoriteLists[index];
                        if (!listItem) {
                            return;
                        }
                        button.dataset.addUrl = listItem.addLink || button.dataset.addUrl || "";
                        button.dataset.removeUrl = listItem.removeLink || button.dataset.removeUrl || "";
                        button.dataset.inList = (listItem.isInFavoriteList === true).toString();
                        var icon = button.querySelector("i");
                        if (icon) {
                            icon.classList.toggle("fas", listItem.isInFavoriteList === true);
                            icon.classList.toggle("fal", listItem.isInFavoriteList !== true);
                        }
                    });
                });
            }

            function renderProductData(data, requestUrl, isCommitted) {
                var product = getProductFromData(data);
                if (!product) {
                    return;
                }
                renderHandlebarsSection("PriceAndActions", data);
                renderHandlebarsSection("PriceAndActions_quantity", data);
                renderHandlebarsSection("Price", data);
                updateGallery(product);
                if (isCommitted) {
                    renderHandlebarsSection("PriceAndActions", data);
                    updateVariantLabels(product);
                }
                if (isCommitted) {
                    state.currentVariantId = product.variantid || "";
                    state.currentUnitId = product.unitId || state.currentUnitId || "";
                    state.previewVariantId = "";
                    state.previewUnitId = "";
                    setCurrentFeedUrl(requestUrl);
                    updateVariantHiddenInputs(product);
                    updateVariantLinkedUrls(product);
                    updateFavorites(product);
                    trackVariantViewEvents(product);
                    syncVariantSelections(state.currentVariantId);
                    if (product.priceDouble && typeof updatePaymentOptions === "function") {
                        updatePaymentOptions(product.priceDouble, product.currencySymbol || "");
                    }
                    if (product.link) {
                        if (state.hasAppliedInitialTrackedUrl) {
                            history.replaceState({ variantId: state.currentVariantId }, "", product.link);
                        }
                        state.hasAppliedInitialTrackedUrl = true;
                    }
                }
            }
            function previewVariant(variantId) {
                if (!variantId || variantId === state.currentVariantId) {
                    clearPreview();
                    return;
                }
                state.previewVariantId = variantId;
                state.previewUnitId = state.currentUnitId || "";
                getProductData(variantId, state.previewUnitId, buildFeedUrl(variantId, state.previewUnitId), function (data) {
                    if (state.previewVariantId === variantId) {
                        var previewData = normalizePreviewPricingData(data);
                        renderProductData(previewData, buildFeedUrl(variantId, state.previewUnitId), false);
                    }
                });

            function clearPreview() {
                if (!state.previewVariantId) {
                    return;
                }
                state.previewVariantId = "";
                state.previewUnitId = "";
                if (state.currentVariantId || state.currentFeedUrl) {
                    getProductData(state.currentVariantId, state.currentUnitId, state.currentFeedUrl, function (data, url) {
                        renderProductData(data, url, false);
                    });
                }
            }
            function getGroupIndexForOption(container, optionElement) {
                var variantsByGroup = JSON.parse((container.getAttribute("data-variants") || "[]").replace(/\'/g, "\""));
                var optionId = optionElement.getAttribute("data-variant-id");
                for (var groupIndex = 0; groupIndex < variantsByGroup.length; groupIndex++) {
                    if (variantsByGroup[groupIndex].indexOf(optionId) !== -1) {
                        return groupIndex;
                    }
                }
                return -1;
            }
            function getPreviewVariantId(optionElement) {
                var container = optionElement.closest(".js-variants");
                if (!container) {
                    return "";
                }
                var totalGroups = parseInt(container.getAttribute("data-total-variant-groups"), 10);
                var selections = state.currentVariantId ? getSelectionsFromVariantId(state.currentVariantId) : getVariantStateFromContainer(container);
                var groupIndex = getGroupIndexForOption(container, optionElement);
                var variantCombinations = JSON.parse((container.getAttribute("data-combinations") || "[]").replace(/\'/g, "\""));
                if (groupIndex === -1) {
                    return "";
                }
                selections[groupIndex] = optionElement.getAttribute("data-variant-id");
                if (selections.filter(Boolean).length !== totalGroups) {
                    return "";
                }
                var previewVariantId = buildVariantIdFromSelections(selections);
                var hasCombination = variantCombinations.some(function (combination) {
                    return buildVariantIdFromSelections(combination) === previewVariantId;
                });
                return hasCombination ? previewVariantId : "";
            }
            function commitCurrentSelection(container) {
                var sortedSelections = getSortedSelections(container);
                var totalGroups = parseInt(container.getAttribute("data-total-variant-groups"), 10);
                if (sortedSelections.length !== totalGroups) {
                    if (state.currentVariantId) {
                        syncVariantSelections(state.currentVariantId);
                    }
                    return;
                }
                var selectedVariantId = buildVariantIdFromSelections(sortedSelections);
                var requestUrl = buildFeedUrl(selectedVariantId, state.currentUnitId);
                getProductData(selectedVariantId, state.currentUnitId, requestUrl, function (data, url) {
                    renderProductData(data, url, true);
                });
            }
            function bindVariantInteractions() {
                getProductContainers().forEach(function (container) {
                    Array.prototype.slice.call(container.querySelectorAll(".js-variant-option")).forEach(function (option) {
                        option.addEventListener("click", function () {
                            window.setTimeout(function () {
                                commitCurrentSelection(container);
                            }, 0);
                        });
                        option.addEventListener("mouseenter", function () {
                            previewVariant(getPreviewVariantId(option));
                        });
                        option.addEventListener("mouseleave", function () {
                            clearPreview();
                        });
                        option.addEventListener("focus", function () {
                            previewVariant(getPreviewVariantId(option));
                        });
                        option.addEventListener("blur", function () {
                            clearPreview();
                        });
                    });
                    Array.prototype.slice.call(container.querySelectorAll("select[id^='VariantSelector_']")).forEach(function (select) {
                        select.addEventListener("change", function () {
                            window.setTimeout(function () {
                                commitCurrentSelection(container);
                            }, 0);
                        });
                    });
                });
            }
            function prefetchVariantData() {
                if (state.prefetchStarted) {
                    return;
                }
                var primaryContainer = getPrimaryVariantsContainer();
                if (!primaryContainer) {
                    return;
                }
                state.prefetchStarted = true;
                var variantCombinations = JSON.parse((primaryContainer.getAttribute("data-combinations") || "[]").replace(/\'/g, "\""));
                variantCombinations.forEach(function (combination) {
                    var variantId = buildVariantIdFromSelections(combination);
                    if (!variantId || variantId === state.currentVariantId) {
                        return;
                    }
                    getProductData(variantId, state.currentUnitId, buildFeedUrl(variantId, state.currentUnitId), function () { });
                });
            }
            function selectUnit(unitId, baseLink) {
                state.currentUnitId = unitId || "";
                state.prefetchStarted = false;
                var activeVariantId = state.previewVariantId || state.currentVariantId;
                var requestUrl = buildFeedUrl(activeVariantId, state.currentUnitId, baseLink);
                getProductData(activeVariantId, state.currentUnitId, requestUrl, function (data, url) {
                    renderProductData(data, url, true);
                    prefetchVariantData();
                });
            }
            function init() {
                if (state.initialized) {
                    return;
                }
                state.initialized = true;
                bindVariantInteractions();
                getProductData(state.currentVariantId, state.currentUnitId, state.currentFeedUrl, function (data, url) {
                    var product = getProductFromData(data);
                    if (product) {
                        state.currentVariantId = product.variantid || state.currentVariantId || "";
                        state.currentUnitId = product.unitId || "";
                        state.currentFeedUrl = url;
                        renderProductData(data, url, true);
                        prefetchVariantData();
                    }
                });
            }
            return {
                Init: init,
                ClearPreview: clearPreview,
                SelectUnit: selectUnit,
                GetCurrentUnitId: function () {
                    return state.currentUnitId || "";
                }
            };

    <script>
        document.addEventListener("DOMContentLoaded", function () {
            document.addEventListener("click", function (e) {
                const btn = e.target.closest(".quick-buy");
                if (!btn) return;
                e.preventDefault();
                const p = btn.getAttribute("data-product-id");
                const h = btn.getAttribute("data-options-header");
                btn.classList.add("registerbutton--loading");
                document.querySelector("#buymodal .modal__body").innerHTML = "";
                fetch("/btapi/quick-buy/product/v2/" + p)
                    .then(r => r.json())
                    .then(response => {
                        document.querySelector("#buymodal .modal__body").innerHTML = response;
                        document.querySelector("#buymodal .buy-modal-header").innerHTML = h;
                        document.getElementById("BuyModalTrigger").checked = true;
                        btn.classList.remove("registerbutton--loading");
                    })
                    .catch(() => {
                        btn.classList.remove("registerbutton--loading");
                    });
            });
            document.getElementById("buymodalAddToCart").addEventListener("click", function (event) {
                const e = document.querySelector("#buymodal .buy-modal-option.buy-option-selected");
                if (!e) return;
                const pid = e.getAttribute("data-product-id");
                const vid = e.getAttribute("data-variant-id");
                const uid = e.getAttribute("data-unit-id");
                const info = e.getAttribute("data-product-info");
                Cart.AddToCart(event, {
                    id: pid,
                    variantId: vid,
                    unitId: uid,
                    productInfo: info,
                    quantity: 1
                });
                document.getElementById("BuyModalTrigger").checked = false;
                AddedToCartConfirmation(1);
            });
            document.getElementById("BuyModalTrigger").addEventListener("change", function () {
                if (!this.checked) {
                    document.querySelector("#buymodal .modal__body").innerHTML = "";
                }
            });
        });
        function selectOption(el) {
            document.querySelectorAll(".buy-option-selected").forEach(d => {
                d.classList.remove("buy-option-selected");
            });
            el.classList.add("buy-option-selected");
        }
        function AddedToCartConfirmation(seconds) {
            const modalCheckbox = document.getElementById("AddedToCartTrigger");
            modalCheckbox.checked = true;
            setTimeout(function () {
                modalCheckbox.checked = false;
            }, seconds * 1000);
        }
        function AddedToCart() {
            document.getElementById("AddedToCartTrigger").checked = true;
        }
